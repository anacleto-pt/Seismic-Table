int sismo_W_len = 2673;

int indiceAtual = 0;
bool sismoLigado = false;
unsigned long ultimoUpdate = 0;
unsigned long tempoInicioSismo = 0;

void setup() {
  Serial.begin(115200);
  
  // Configuração do pino Enable
  pinMode(pinoEnable, OUTPUT);
  
  // LIMITES DE SEGURANÇA (Tetos de processamento altos para a biblioteca não prender)
  motorN.setMaxSpeed(60000);
  motorE.setMaxSpeed(60000);
  motorS.setMaxSpeed(60000);
  motorW.setMaxSpeed(60000);
  
  Serial.println(">>> ARRANQUE IMEDIATO DA SIMULAÇÃO <<<");
  
  // ========================================================
  // AS 4 LINHAS QUE FAZEM O MOTOR ARRANCAR LOGO SEM ESPERAR:
  digitalWrite(pinoEnable, LOW); // Ativa imediatamente a energia nos drivers
  delay(50);                     // Margem de milissegundos para estabilizar a eletrónica
  sismoLigado = true;            // Ativa a flag do loop
  tempoInicioSismo = millis();   // Define o t=0 para a contagem do sismo
  ultimoUpdate = millis();       // Sincroniza a janela de 20ms
  // ========================================================
}

void loop() {
  // Execução síncrona rígida a cada 20ms
  if (sismoLigado) {
    if (millis() - ultimoUpdate >= 20) {
      ultimoUpdate = millis();

      if (indiceAtual < totalPontos) {
        float tempoAtual = (millis() - tempoInicioSismo) / 1000.0;
        
        // Conversão de Passos/Janela para Passos/Segundo: Velocidade = (passos) / 0.02s
        float velN = (float)sismo_N[indiceAtual] / 0.02;
        float velE = (float)sismo_E[indiceAtual] / 0.02;
        float velS = (float)sismo_S[indiceAtual] / 0.02;
        float velW = (float)sismo_W[indiceAtual] / 0.02;
        
        // Atualiza a velocidade instantânea de cada motor
        motorN.setSpeed(velN);
        motorE.setSpeed(velE);
        motorS.setSpeed(velS);
        motorW.setSpeed(velW);
        
        // Telemetria via Serial (Apenas para monitorização)
        Serial.print(tempoAtual, 3); Serial.print(",");
        Serial.print(velN, 0);       Serial.print(",");
        Serial.print(velE, 0);       Serial.print(",");
        Serial.print(velS, 0);       Serial.println(velW, 0);

        indiceAtual++;
      } 
      else {
        // O sismo acabou: desliga as saídas e corta a corrente aos motores
        sismoLigado = false;
        motorN.setSpeed(0); motorE.setSpeed(0);
        motorS.setSpeed(0); motorW.setSpeed(0);
        
        digitalWrite(pinoEnable, HIGH); // Standby térmico (motores relaxam e arrefecem)
        Serial.println("--- SIMULAÇÃO SÍSMICA TERMINADA ---");
      }
    }
    
    // Mantém os motores a pulsar continuamente entre as janelas de tempo
    motorN.runSpeed();
    motorE.runSpeed();
    motorS.runSpeed();
    motorW.runSpeed();
  }
}
