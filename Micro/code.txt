bool sismoLigado = false;
int indiceAtual = 0;
unsigned long ultimoUpdate = 0;
unsigned long tempoInicioSismo = 0;

void setup() {
  Serial.begin(115200);
  pinMode(pinoEnable, OUTPUT);

  motorN.setMaxSpeed(60000);
  motorE.setMaxSpeed(60000);
  motorS.setMaxSpeed(60000);
  motorW.setMaxSpeed(60000);

  digitalWrite(pinoEnable, HIGH);

  Serial.println("Envia qualquer tecla + Enter para iniciar simulação sísmica...");
}

void loop() {
  if (!sismoLigado) {
    if (Serial.available() > 0) {
      while (Serial.available()) Serial.read();
      digitalWrite(pinoEnable, LOW);
      delay(50);
      indiceAtual = 0;
      sismoLigado = true;
      tempoInicioSismo = millis();
      ultimoUpdate = millis();
      Serial.println("tempo_s,velN,velE,velS,velW");
      Serial.println(">>> INÍCIO DA SIMULAÇÃO SÍSMICA <<<");
    }
    return;
  }

  if (millis() - ultimoUpdate >= 40) {
    ultimoUpdate = millis();

    if (indiceAtual < totalPontos) {
      float tempoAtual = (millis() - tempoInicioSismo) / 1000.0;

      float velN = (float)sismo_N[indiceAtual] / 0.04;
      float velE = (float)sismo_E[indiceAtual] / 0.04;
      float velS = (float)sismo_S[indiceAtual] / 0.04;
      float velW = (float)sismo_W[indiceAtual] / 0.04;

      motorN.setSpeed(velN);
      motorE.setSpeed(velE);
      motorS.setSpeed(velS);
      motorW.setSpeed(velW);

      Serial.print(tempoAtual, 3); Serial.print(",");
      Serial.print(velN, 0);       Serial.print(",");
      Serial.print(velE, 0);       Serial.print(",");
      Serial.print(velS, 0);       Serial.print(",");
      Serial.println(velW, 0);

      indiceAtual++;
    } else {
      sismoLigado = false;
      motorN.setSpeed(0); motorE.setSpeed(0);
      motorS.setSpeed(0); motorW.setSpeed(0);
      digitalWrite(pinoEnable, HIGH);
      Serial.println("--- SIMULAÇÃO SÍSMICA TERMINADA ---");
    }
  }

  motorN.runSpeed();
  motorE.runSpeed();
  motorS.runSpeed();
  motorW.runSpeed();
}
